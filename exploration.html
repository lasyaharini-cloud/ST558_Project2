Plot 7: Average Housing Prices by Region and Type

This heatmap shows the average housing price for each property type across regions.

Code 
# Calculate average prices
avg_prices <- housing_clean |>
  group_by(Regionname, Type) |>
  summarise(AvgPrice = mean(Price, na.rm = TRUE),
            .groups = "drop")

# Heatmap
ggplot(avg_prices, aes(x = Type, y = Regionname, fill = AvgPrice)) +
  geom_tile(color = "white") +
  labs(title = "Average Housing Prices by Region and Type",
       x = "Property Type",
       y = "Region",
       fill = "Average Price") +
  theme_minimal()
